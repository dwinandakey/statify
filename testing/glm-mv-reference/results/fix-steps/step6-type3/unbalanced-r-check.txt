--- R script
suppressMessages(library(car))
d <- read.csv("testing/glm-mv-reference/data/two-way manova.csv")
d <- d[-c(4, 11, 12, 21, 26, 31), ]
d$A <- factor(d$faktorA); d$B <- factor(d$faktorB)
options(contrasts = c("contr.sum", "contr.poly"))
for (y in c("Y1A1", "Y2A1")) {
  m <- lm(as.formula(paste(y, "~ A * B")), data = d)
  a3 <- Anova(m, type = 3); a2 <- Anova(m, type = 2)
  for (t in c("(Intercept)", "A", "B", "A:B")) cat(sprintf("R typeIII %s %s %.15g\n", y, t, a3[t, "Sum Sq"]))
  for (t in c("A", "B", "A:B")) cat(sprintf("R typeII %s %s %.15g\n", y, t, a2[t, "Sum Sq"]))
  cat(sprintf("R Total %s %.15g n=%d\n", y, sum(d[[y]]^2), nrow(d)))
}
