m <- lm(cbind(Y1A1, Y2A1) ~ A + B, data = d)
st <- function(H, E) { ev <- Re(eigen(solve(E) %*% H)$values); c(sum(ev/(1+ev)), prod(1/(1+ev)), sum(ev), max(ev)) }
a3 <- Anova(m, type = 3)
for (t in c("(Intercept)", "A", "B")) { s <- st(a3$SSP[[t]], a3$SSPE); cat(sprintf("R MV %s | Pillai %.15g Wilks %.15g Hotelling %.15g Roy %.15g\n", t, s[1], s[2], s[3], s[4])) }
