--- R script
suppressMessages(library(car))
d <- read.csv("data/two-way manova tak seimbang.csv")
d$A <- factor(d$faktorA); d$B <- factor(d$faktorB)
options(contrasts = c("contr.sum", "contr.poly"))
m <- lm(cbind(Y1A1, Y2A1) ~ A * B, data = d)
st <- function(H, E) { ev <- Re(eigen(solve(E) %*% H)$values); c(sum(ev/(1+ev)), prod(1/(1+ev)), sum(ev), max(ev)) }
E <- crossprod(residuals(m))
# Type I: sequential SSCP from the QR effects (A, B|A, A:B|A,B)
eff <- qr.qty(m$qr, cbind(d$Y1A1, d$Y2A1)); asn <- m$assign[m$qr$pivot[1:m$rank]]
for (k in 1:3) { H <- crossprod(eff[seq_along(asn), , drop = FALSE][asn == k, , drop = FALSE]); cat(sprintf("R typeI %s Pillai %.15g Wilks %.15g Hotelling %.15g Roy %.15g\n", c("A","B","A:B")[k], st(H,E)[1], st(H,E)[2], st(H,E)[3], st(H,E)[4])) }
a2 <- Anova(m, type = 2)
for (t in c("A","B","A:B")) { s <- st(a2$SSP[[t]], a2$SSPE); cat(sprintf("R typeII %s Pillai %.15g Wilks %.15g Hotelling %.15g Roy %.15g\n", t, s[1], s[2], s[3], s[4])) }
a3 <- Anova(m, type = 3)
for (t in c("A","B","A:B")) { s <- st(a3$SSP[[t]], a3$SSPE); cat(sprintf("R typeIII %s Pillai %.15g Wilks %.15g Hotelling %.15g Roy %.15g\n", t, s[1], s[2], s[3], s[4])) }

--- Statify script (node)
import fs from "fs";
const MV = "D:/0.POLTSTAT STIS/Tugas Kuliah/Skripsi/topik baru statify/statify64/frontend/components/Modals/Analyze/general-linear-model/multivariate";
const { default: init, MultivariateAnalysis } = await import("file:///" + MV + "/rust/pkg/wasm.js");
await init({ module_or_path: fs.readFileSync(MV + "/rust/pkg/wasm_bg.wasm") });
const fx = JSON.parse(fs.readFileSync(MV + "/__test__/fixtures/mv-reference-values.json", "utf8"));
const p = fx.configs.mv6.payload;
for (const t of ["typeI", "typeII", "typeIII"]) {
  const cfg = structuredClone(p.config_data); cfg.model.SumOfSquareMethod = t;
  const a = new MultivariateAnalysis(p.dep_data, p.fix_factor_data, p.covar_data, p.wls_data, p.dep_data_defs, p.fix_factor_data_defs, p.covar_data_defs, p.wls_data_defs, cfg);
  const plain = (x) => x instanceof Map ? Object.fromEntries([...x].map(([k, v]) => [k, plain(v)])) : Array.isArray(x) ? x.map(plain) : (x && typeof x === "object") ? Object.fromEntries(Object.entries(x).map(([k, v]) => [k, plain(v)])) : x;
  const r = plain(a.get_results()); const eff = r.multivariate_tests.effects;
  for (const e of ["faktorA", "faktorB", "faktorA * faktorB"]) {
    const s = eff[e] ?? eff[e.replace(/ /g, "")]; if (!s) { console.log(t, "keys", Object.keys(eff)); continue; }
    const v = (k) => (s[k] ?? s.get?.(k))?.value;
    console.log(`${t} ${e} Pillai ${v("Pillai's Trace")} Wilks ${v("Wilks' Lambda")} Hotelling ${v("Hotelling's Trace")} Roy ${v("Roy's Largest Root")}`);
  }
}
