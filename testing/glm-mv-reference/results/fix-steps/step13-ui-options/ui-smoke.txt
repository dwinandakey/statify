--- ui-smoke.cjs
const path = require("path");
const REPO = "D:/0.POLTSTAT STIS/Tugas Kuliah/Skripsi/topik baru statify/statify64";
const { chromium } = require(path.join(REPO, "node_modules/@playwright/test"));
const runner = require(path.join(REPO, "testing/glm-web-worker/experiment/run-experiment.cjs"));
const toastText = async (page) => { await page.waitForTimeout(700); return (await page.locator("[data-sonner-toast]").allInnerTexts()).map((t) => t.split(String.fromCharCode(10)).join(" ")).join(" || "); };
(async () => {
  const b = await chromium.launch(); const out = [];
  const ctx = await b.newContext({ viewport: { width: 1600, height: 1000 } }); const page = await ctx.newPage();
  await page.goto("http://localhost:3101/dashboard/data"); await page.locator('[data-testid="main-navbar"]').waitFor();
  await runner.importCsv(page, path.join(REPO, "testing/glm-rm-reference/data/rm_b.csv"));
  await runner.clearResults(page);
  const openGlm = async (item) => { await page.locator('[data-testid="analyze-menu-trigger"]').click(); await page.getByRole("menuitem", { name: "General Linear Model" }).click(); await page.getByRole("menuitem", { name: item, exact: true }).click(); };
  // MV: Plots disabled, Post Hoc methods
  await openGlm("Multivariate"); await page.locator("#multivariate-ok-button").waitFor();
  out.push(["MV Plots disabled", await page.locator("#multivariate-plots-button").isDisabled()]);
  out.push(["MV Plots note", await page.getByText("Plots are not supported in this version.", { exact: true }).count()]);
  await page.getByRole("button", { name: "Post Hoc", exact: true }).click();
  const ph = page.getByRole("dialog").filter({ has: page.getByText("Multivariate: Post Hoc", { exact: true }) }); await ph.waitFor();
  for (const id of ["Lsd", "Bonfe", "Sidak", "Scheffe", "Tu", "Games", "Dunnett"]) out.push([`MV Post Hoc #${id} disabled`, await ph.locator("#" + id).isDisabled()]);
  out.push(["MV Post Hoc note", await ph.getByText("Only LSD, Bonferroni, and Sidak are supported in this version.", { exact: true }).count()]);
  await ph.getByRole("button", { name: "Cancel", exact: true }).click().catch(async () => { await page.keyboard.press("Escape"); });
  await page.waitForTimeout(300);
  await page.getByRole("button", { name: "Cancel", exact: true }).first().click().catch(() => {});
  await page.waitForTimeout(500);
  // RM: Define without factor / without measure
  await openGlm("Repeated Measures"); await page.locator("#factorName").waitFor();
  await page.getByRole("button", { name: "Define", exact: true }).click();
  out.push(["RM Define tanpa faktor -> toast", await toastText(page)]);
  await page.locator("#factorName").fill("waktu"); await page.locator("#factorLevels").fill("4");
  await page.getByRole("button", { name: "Add", exact: true }).nth(0).click();
  await page.getByRole("button", { name: "Define", exact: true }).click();
  out.push(["RM Define tanpa measure -> toast", await toastText(page)]);
  await page.locator("#factorName").fill("kedua"); await page.locator("#factorLevels").fill("2");
  await page.getByRole("button", { name: "Add", exact: true }).nth(0).click();
  out.push(["RM faktor kedua -> toast", await toastText(page)]);
  await page.locator("#measureName").fill("skor"); await page.getByRole("button", { name: "Add", exact: true }).nth(1).click();
  await page.getByRole("button", { name: "Define", exact: true }).click();
  await page.getByText("Within-Subjects Variables:", { exact: true }).waitFor();
  out.push(["RM Plots disabled", await page.locator("#repeated-measures-plots-button").isDisabled()]);
  out.push(["RM Post Hoc disabled", await page.locator("#repeated-measures-posthoc-button").isDisabled()]);
  out.push(["RM Post Hoc note", await page.getByText("Post Hoc is not supported in this version. Use EM Means > Compare main effects for pairwise comparisons.", { exact: true }).count()]);
  await page.getByRole("button", { name: "OK", exact: true }).click();
  out.push(["RM OK slot kosong -> toast", await toastText(page)]);
  await page.getByRole("button", { name: "Options", exact: true }).click();
  const op = page.getByRole("dialog").last(); await page.locator("#LackOfFit").waitFor();
  for (const id of ["SprVsLevel", "ResPlot", "LackOfFit", "HomogenTest"]) out.push([`RM Options #${id} disabled`, await page.locator("#" + id).isDisabled()]);
  out.push(["RM Options note", await page.getByText("Spread-vs.-level plots, residual plots, and the lack-of-fit test are not supported in this version.", { exact: true }).count()]);
  for (const [k, v] of out) console.log(`${k}: ${JSON.stringify(v)}`);
  await b.close();
})().catch((e) => { console.error("GAGAL", e.message.split(String.fromCharCode(10))[0]); process.exit(1); });
