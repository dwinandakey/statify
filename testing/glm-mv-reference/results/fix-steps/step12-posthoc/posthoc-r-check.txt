--- R script
d <- read.csv("testing/glm-mv-reference/data/one-way manova.csv"); d$t <- factor(d$treatment)
for (y in c("y1", "y2")) {
  f <- lm(as.formula(paste(y, "~ t")), data = d); mse <- sum(resid(f)^2) / f$df.residual; df <- f$df.residual
  m <- tapply(d[[y]], d$t, mean); n <- tapply(d[[y]], d$t, length); L <- levels(d$t); c <- choose(length(L), 2)
  for (i in 1:(length(L)-1)) for (j in (i+1):length(L)) {
    diff <- m[i] - m[j]; se <- sqrt(mse * (1/n[i] + 1/n[j])); p <- 2 * pt(-abs(diff/se), df)
    for (meth in c("LSD", "Bonferroni", "Sidak")) {
      padj <- switch(meth, LSD = p, Bonferroni = min(1, c * p), Sidak = 1 - (1 - p)^c)
      q <- switch(meth, LSD = 0.025, Bonferroni = 0.05 / (2 * c), Sidak = (1 - 0.95^(1/c)) / 2)
      tc <- qt(1 - q, df)
      cat(sprintf("R %s %s %s-%s | diff %.15g SE %.15g Sig %.15g CI %.15g %.15g\n", y, meth, L[i], L[j], diff, se, padj, diff - tc*se, diff + tc*se))
    }
  }
}
